cmake_minimum_required(VERSION 3.20)
project(nightfall VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

target_include_directories(nightfall PUBLIC src)
if(MSVC)
  target_compile_options(nightfall PRIVATE /W4 /permissive-)
else()
  target_compile_options(nightfall PRIVATE -Wall -Wextra -Wpedantic)
endif()

target_link_libraries(nightfall_cli PRIVATE nightfall)
